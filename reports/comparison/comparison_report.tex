\noindent\textbf{Design codes:}
\begin{itemize}
\item \textbf{CLA} — Classical DID (single adoption date, binary treatment)
\item \textbf{SAD} — Staggered Adoption Design (multiple adoption dates, binary)
\item \textbf{SFSD} — Staggered First Switch Design (non-binary/non-absorbing)
\item \textbf{HAD} — Heterogeneous Adoption Design (common adoption, heterogeneous dose)
\item \textbf{SSD} — Switchers and Stayers Design (sketch\_did\_design only)
\item \textbf{Other} — Does not fit the above categories
\end{itemize}
